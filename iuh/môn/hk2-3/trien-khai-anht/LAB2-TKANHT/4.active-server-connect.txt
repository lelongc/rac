# Lấy danh sách kết nối TCP đang hoạt động
$connections = Get-NetTCPConnection | Where-Object { $_.State -eq "Established" }

# Tạo danh sách kết quả
$results = @()

foreach ($conn in $connections) {
    $results += [PSCustomObject]@{
        LocalIP    = $conn.LocalAddress
        LocalPort  = $conn.LocalPort
        RemoteIP   = $conn.RemoteAddress
        RemotePort = $conn.RemotePort
        State      = $conn.State
        PID        = $conn.OwningProcess
    }
}

# Hiển thị kết quả ra màn hình
$results | Format-Table -AutoSize

# Xuất ra file CSV để phân tích thêm
$results | Export-Csv -Path "ActiveConnections.csv" -NoTypeInformation -Encoding UTF8

Write-Host "`n✅ Kết nối đang hoạt động đã được ghi vào ActiveConnections.csv" -ForegroundColor Yellow
