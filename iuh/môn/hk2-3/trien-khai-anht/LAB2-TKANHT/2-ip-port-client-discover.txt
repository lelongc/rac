# Cấu hình dải IP và danh sách cổng cần kiểm tra
$baseIP = "192.168.1"
$start = 1
$end = 254
$ports = @(20, 21, 22, 80, 443)  # Bạn có thể thêm các cổng khác tại đây

# Tạo danh sách kết quả
$results = @()

# Duyệt qua từng IP và từng cổng
for ($i = $start; $i -le $end; $i++) {
    $ip = "$baseIP.$i"
    foreach ($port in $ports) {
        Write-Host "Kiểm tra $ip:$port..." -ForegroundColor Cyan

        $check = Test-NetConnection -ComputerName $ip -Port $port -InformationLevel Quiet

        $status = if ($check) { "Open" } else { "Closed" }

        # Ghi kết quả vào danh sách
        $results += [PSCustomObject]@{
            IPAddress = $ip
            Port      = $port
            Status    = $status
            Timestamp = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
        }

        # Hiển thị kết quả
        if ($check) {
            Write-Host "$ip:$port mở." -ForegroundColor Green
        } else {
            Write-Host "$ip:$port đóng hoặc không phản hồi." -ForegroundColor Red
        }
    }
}

# Xuất kết quả ra file CSV
$results | Export-Csv -Path "PortScanResults.csv" -NoTypeInformation -Encoding UTF8

Write-Host "`n✅ Quét hoàn tất. Kết quả đã lưu vào PortScanResults.csv" -ForegroundColor Yellow
