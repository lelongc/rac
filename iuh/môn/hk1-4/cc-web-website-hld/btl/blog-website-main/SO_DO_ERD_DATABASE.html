    blogs ||--o{ comments : "has"

    blogs ||--o{ blog_views : "viewed in"

    tags ||--o{ blog_tags : "assigned to"

    comments ||--o{ comments : "replies to (parent_id)"

    %% ==========================================
    %% ĐỊNH NGHĨA CẤU TRÚC CHI TIẾT CÁC BẢNG
    %% ==========================================

    users {
        binary(16) id PK "Mã định danh User (UUID)"
        varchar(50) username UK "Tên đăng nhập duy nhất"
        varchar(255) email UK "Địa chỉ email"
        varchar(255) password "Mật khẩu mã hóa BCrypt"
        varchar(500) avatar_url "Đường dẫn ảnh đại diện"
        varchar(255) bio "Tiểu sử giới thiệu"
        datetime created_at "Ngày tạo tài khoản"
    }

    user_roles {
        binary(16) user_id PK,FK "Tham chiếu users(id)"
        varchar(50) role PK "Tên quyền: ROLE_ADMIN, ROLE_USER"
    }

    categories {
        binary(16) id PK "Mã danh mục (UUID)"
        varchar(100) name UK "Tên chủ đề bài viết"
        varchar(255) description "Mô tả chuyên mục"
    }

    tags {
        binary(16) id PK "Mã thẻ (UUID)"
        varchar(50) name UK "Tên thẻ tag: Java, SpringBoot..."
    }

    blogs {
        binary(16) id PK "Mã bài viết (UUID)"
        varchar(255) title "Tiêu đề bài viết"
        longtext content "Nội dung bài viết đầy đủ"
        varchar(500) description "Tóm tắt ngắn bài viết"
        varchar(500) banner "URL ảnh bìa Cloudinary"
        int total_reads "Số lượt xem (Views)"
        datetime created_at "Thời điểm đăng bài"
        datetime updated_at "Thời điểm cập nhật"
        binary(16) author_id FK "Tác giả -> users(id)"
        binary(16) category_id FK "Danh mục -> categories(id)"
    }

    blog_tags {
        binary(16) blog_id PK,FK "Tham chiếu blogs(id)"
        binary(16) tag_id PK,FK "Tham chiếu tags(id)"
    }

    comments {
        binary(16) id PK "Mã bình luận (UUID)"
        text content "Nội dung comment"
        datetime created_at "Thời gian bình luận"
        binary(16) user_id FK "Người bình luận -> users(id)"
        binary(16) blog_id FK "Bài viết -> blogs(id)"
        binary(16) parent_id FK "Bình luận cha (Reply) -> comments(id)"
    }

    blog_likes {
        binary(16) id PK "Mã lượt thích (UUID)"
        datetime created_at "Thời gian bấm tim"
        binary(16) user_id FK "Người thích -> users(id)"
        binary(16) blog_id FK "Bài được thích -> blogs(id)"
    }

    bookmarks {
        binary(16) id PK "Mã lưu trữ (UUID)"
        datetime created_at "Thời điểm lưu bài"
        binary(16) user_id FK "Người lưu -> users(id)"
        binary(16) blog_id FK "Bài được lưu -> blogs(id)"
    }

    user_follows {
        binary(16) id PK "Mã theo dõi (UUID)"
        datetime created_at "Thời điểm theo dõi"
        binary(16) follower_id FK "Người theo dõi -> users(id)"
        binary(16) following_id FK "Người được theo dõi -> users(id)"
    }

    notifications {
        binary(16) id PK "Mã thông báo (UUID)"
        varchar(500) content "Nội dung thông báo tương tác"
        boolean is_read "Trạng thái đã đọc (true/false)"
        datetime created_at "Thời điểm phát thông báo"
        binary(16) user_id FK "Người nhận thông báo -> users(id)"
    }

    blog_views {
        binary(16) id PK "Mã lượt đọc (UUID)"
        datetime viewed_at "Thời điểm đọc bài"
        binary(16) blog_id FK "Bài viết -> blogs(id)"
        binary(16) user_id FK "Người xem -> users(id)"
    }
