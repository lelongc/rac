=====================================================================
# CÁC LỆNH CẤU HÌNH TUẦN 5 (V2) - BÀI 1 (ACCESS CONTROL LIST)
=====================================================================

---------------------------------------------------------------------
1. THIẾT BỊ: SW3 (MULTILAYER SWITCH - ROUTING & ACL CORE)
---------------------------------------------------------------------
enable
configure terminal
ip routing

# LƯU Ý QUAN TRỌNG: Tắt CEF chống lỗi giả lập IOL trong EVE-NG. 
# Nếu không tắt, switch ảo sẽ không chuyển tiếp (route) gói tin giữa các VLAN 
# mặc dù bảng định tuyến và ARP hiển thị đầy đủ, dẫn đến ping bị timeout.
no ip cef

vlan 2
vlan 3
exit
interface vlan 1
 ip address 192.168.1.1 255.255.255.0
 no shutdown
 exit
interface vlan 2
 ip address 172.16.2.1 255.255.255.0
 no shutdown
 exit
interface vlan 3
 ip address 172.16.3.1 255.255.255.0
 no shutdown
 exit

# Cấu hình cổng e0/1 làm cổng Access VLAN 1 nối với Switch0
interface e0/1
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

# Cấu hình các cổng e0/2, e0/3 làm cổng Trunk nối sang Switch1 và Switch2
interface range e0/2 - 3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
 exit

# Cấu hình định tuyến RIP
router rip
 version 2
 no auto-summary
 network 192.168.1.0
 network 172.16.0.0
 exit

# Cấu hình Access Control List (ACL) cho VLAN 3
# - Cho phép PC_FTP (172.16.3.100) kết nối FTP tới Server0 (192.168.1.2)
# - Cho phép PC_FTP (172.16.3.100) PING tới Server0 (ICMP) để kiểm tra thực tế (vì VPCS không có trình duyệt/client FTP thực sự).
# - Chặn toàn bộ phần còn lại của VLAN 3 (ví dụ PC0 172.16.3.2) tới Server0.
ip access-list extended VLAN3_ACL
 permit tcp host 172.16.3.100 host 192.168.1.2 eq ftp
 permit tcp host 172.16.3.100 host 192.168.1.2 eq ftp-data
 permit icmp host 172.16.3.100 host 192.168.1.2
 deny ip 172.16.3.0 0.0.0.255 host 192.168.1.2
 permit ip any any
 exit
interface vlan 3
 ip access-group VLAN3_ACL in
 exit

# Cấu hình Access Control List (ACL) cho VLAN 2
# Chặn một PC (172.16.2.100 - PC_BLOCKED) đến Server1 (192.168.1.3), còn lại cho phép đi
ip access-list extended VLAN2_ACL
 deny ip host 172.16.2.100 host 192.168.1.3
 permit ip any any
 exit
interface vlan 2
 ip access-group VLAN2_ACL in
 end
write memory


---------------------------------------------------------------------
2. THIẾT BỊ: Switch0 (VLAN 1 ACCESS SWITCH)
---------------------------------------------------------------------
enable
configure terminal
# Bật và cấu hình chế độ Access VLAN 1 cho tất cả các cổng nối SW3 (e0/1) và Servers (e0/2, e0/3)
interface range e0/1 - 3
 switchport mode access
 switchport access vlan 1
 no shutdown
 end
write memory


---------------------------------------------------------------------
3. THIẾT BỊ: Switch1 (VLAN 3 ACCESS SWITCH)
---------------------------------------------------------------------
enable
configure terminal
vlan 3
exit
interface e0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
 exit
interface range e0/2 - 3
 switchport mode access
 switchport access vlan 3
 no shutdown
 end
write memory


---------------------------------------------------------------------
4. THIẾT BỊ: Switch2 (VLAN 2 ACCESS SWITCH)
---------------------------------------------------------------------
enable
configure terminal
vlan 2
exit
interface e0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
 exit
interface range e0/2 - 3
 switchport mode access
 switchport access vlan 2
 no shutdown
 end
write memory


---------------------------------------------------------------------
5. CẤU HÌNH IP TRÊN CÁC MÁY TÍNH (VPCs)
---------------------------------------------------------------------
(Mở console từng VPC và gõ lệnh IP tương ứng dưới đây).

Vùng VLAN 1:
Server0:
ip 192.168.1.2 255.255.255.0 192.168.1.1
save

Server1:
ip 192.168.1.3 255.255.255.0 192.168.1.1
save

Vùng VLAN 3:
PC0 (Máy trạm thường - Bị chặn đến Server0):
ip 172.16.3.2 255.255.255.0 172.16.3.1
save

PC_FTP (Máy trạm đặc quyền - Thông FTP và Ping đến Server0):
ip 172.16.3.100 255.255.255.0 172.16.3.1
save

Vùng VLAN 2:
PC1 (Máy trạm thường - Thông mạng sang Server1):
ip 172.16.2.2 255.255.255.0 172.16.2.1
save

PC_BLOCKED (Máy trạm bị chặn - Bị chặn đến Server1):
ip 172.16.2.100 255.255.255.0 172.16.2.1
save
