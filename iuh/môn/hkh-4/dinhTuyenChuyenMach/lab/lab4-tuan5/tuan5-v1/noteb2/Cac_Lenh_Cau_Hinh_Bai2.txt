# CÁC LỆNH CẤU HÌNH TUẦN 5 - BÀI 2 (ROUTING VLAN / INTER-VLAN ROUTING)

THÔNG TIN SINH VIÊN:
- Tên: Long
- MSSV: 23630851

---------------------------------------------------------------------
1. THIẾT BỊ: SW_SERVER (VTP SERVER)
---------------------------------------------------------------------
enable
configure terminal
vtp domain Long
vtp password 23630851
vtp version 2
vtp mode server
vlan 2
 name IT
vlan 3
 name IS
vlan 4
 name SE
vlan 5
 name CS
interface range e0/0 - 3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
end
write memory

4. THIẾT BỊ: R1 (ROUTER ĐỊNH TUYẾN CHÉO VLAN - ROUTER ON A STICK)
---------------------------------------------------------------------
enable
configure terminal
interface s1/0
 ip address 10.10.10.1 255.255.255.0
 no shutdown
 exit
interface e0/0
 no shutdown
 exit
interface e0/0.2
 encapsulation dot1Q 2
 ip address 192.168.2.1 255.255.255.0
 exit
interface e0/0.3
 encapsulation dot1Q 3
 ip address 192.168.3.1 255.255.255.0
 exit
interface e0/0.4
 encapsulation dot1Q 4
 ip address 192.168.4.1 255.255.255.0
 exit
interface e0/0.5
 encapsulation dot1Q 5
 ip address 192.168.5.1 255.255.255.0
 exit
ip route 172.16.0.0 255.255.255.0 10.10.10.2
end
write memory


---------------------------------------------------------------------
5. THIẾT BỊ: R2 (ROUTER NỐI VỚI HOST NGOÀI)

 no shutdown
