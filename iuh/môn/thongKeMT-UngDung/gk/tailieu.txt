# 7.1 PHÂN PHỐI NHỊ THỨC
print("\n=== PHÂN PHỐI NHỊ THỨC ===")
n = 50  # Số thử nghiệm
p = 0.5  # Xác suất thành công

# Tính xác suất
prob_exact = binom.pmf(25, n, p)  # P(X = 25)
prob_less_equal = binom.cdf(25, n, p)  # P(X ≤ 25)
prob_greater = 1 - binom.cdf(24, n, p)  # P(X ≥ 25)

print(f"P(X = 25) = {prob_exact:.6f}")
print(f"P(X ≤ 25) = {prob_less_equal:.6f}")
print(f"P(X ≥ 25) = {prob_greater:.6f}")

# Kỳ vọng và phương sai
E_X = n * p
Var_X = n * p * (1 - p)
print(f"E(X) = {E_X}")
print(f"Var(X) = {Var_X}")

# 7.2 PHÂN PHỐI POISSON
print("\n=== PHÂN PHỐI POISSON ===")
lambda_val = 3  # Trung bình

# Tính xác suất
prob_0 = poisson.pmf(0, lambda_val)
prob_1 = poisson.pmf(1, lambda_val)
prob_2_or_more = 1 - poisson.cdf(1, lambda_val)

print(f"P(X = 0) = {prob_0:.6f}")
print(f"P(X = 1) = {prob_1:.6f}")
print(f"P(X ≥ 2) = {prob_2_or_more:.6f}")

# 7.3 PHÂN PHỐI CHUẨN
print("\n=== PHÂN PHỐI CHUẨN ===")
mu = 170  # Trung bình
sigma = 10  # Độ lệch chuẩn

# Tính xác suất
prob_less_160 = norm.cdf(160, mu, sigma)
prob_between = norm.cdf(180, mu, sigma) - norm.cdf(160, mu, sigma)
prob_greater_180 = 1 - norm.cdf(180, mu, sigma)

print(f"P(X < 160) = {prob_less_160:.6f}")
print(f"P(160 < X < 180) = {prob_between:.6f}")
print(f"P(X > 180) = {prob_greater_180:.6f}")

# Tìm phân vị
percentile_95 = norm.ppf(0.95, mu, sigma)
print(f"Phân vị 95%: {percentile_95:.2f}")

def quick_stats(series):
    """Tính toán thống kê nhanh cho một cột"""
    return {
        'mean': series.mean(),
        'median': series.median(),
        'std': series.std(),
        'min': series.min(),
        'max': series.max(),
        'q25': series.quantile(0.25),
        'q75': series.quantile(0.75)
    }

def compare_groups(df, group_col, value_col):
    """So sánh thống kê giữa các nhóm"""
    return df.groupby(group_col)[value_col].agg(['mean', 'median', 'std', 'count'])

def plot_distribution(series, title="Phân phối"):
    """Vẽ histogram và boxplot cho một cột"""
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    
    # Histogram
    axes[0].hist(series.dropna(), bins=20, alpha=0.7, edgecolor='black')
    axes[0].set_title(f'Histogram - {title}')
    axes[0].set_xlabel('Giá trị')
    axes[0].set_ylabel('Tần suất')
    
    # Boxplot
    axes[1].boxplot(series.dropna())
    axes[1].set_title(f'Boxplot - {title}')
    axes[1].set_ylabel('Giá trị')
    
    plt.tight_layout()
    plt.show()
